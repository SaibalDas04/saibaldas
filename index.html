managed implementation in class zbp_r_po_processor unique;
strict ( 2 );
with draft;

define behavior for ZR_PurchaseOrder_TP alias PurchaseOrder
persistent table zpo_header_d
draft table zpo_draft_h
lock master total etag LastChangedAt
authorization master ( global )
{
  field ( readonly, numbering : managed ) POUUID;
  field ( mandatory ) Supplier, PurchaseOrg, TargetValue;

  create; update; delete;

  /* Clean Core Compliance: Validation trigger & event emission */
  validation validateSupplier on save { field Supplier; create; }
  determination onOrderCreated on modify { create; }

  draft action Edit;
  draft action Activate optimized;
  draft action Discard;
  draft determine action Prepare { validation validateSupplier; }
}
METHOD validateSupplier.
  " Read business object data via Entity Manipulation Language (EML)
  READ ENTITIES OF ZR_PurchaseOrder_TP IN LOCAL MODE
    ENTITY PurchaseOrder
    FIELDS ( Supplier TargetValue )
    WITH CORRESPONDING #( keys )
    RESULT DATA(lt_po_data).

  LOOP AT lt_po_data INTO DATA(ls_po).
    IF ls_po-TargetValue <= 0.
      " Populate the RAP reported & failed response structures
      APPEND VALUE #( %tky = ls_po-%tky ) TO failed-purchaseorder.
      APPEND VALUE #( %tky = ls_po-%tky
                      %msg = new_message( id       = 'ZPO_MSG'
                                         number   = 001
                                         severity = if_abap_behv_message=>severity-error )
                      %element-TargetValue = if_abap_behv=>mk-on ) TO reported-purchaseorder.
    ENDIF.
  ENDLOOP.
ENDMETHOD.
@AccessControl.authorizationCheck: #CHECK
@EndUserText.label: 'Clean Core Projection View for Purchase Orders'
@Metadata.allowExtensions: true
define root view entity ZC_PurchaseOrderProjection
  provider contract transactional_query
  as projection on ZR_PurchaseOrder_TP
{
  key POUUID,
      PurchaseOrderNumber,
      Supplier,
      PurchaseOrg,
      @Semantics.amount.currencyCode: 'CurrencyCode'
      TargetValue,
      CurrencyCode,
      OverallStatus,
      /* Tier-1 released association resolution */
      _Supplier.SupplierName as SupplierName
}
